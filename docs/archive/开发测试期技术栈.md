# 表格图片识别导出Excel网站 - 开发测试期技术栈

## 1. 概述

本文档专为开发和测试阶段设计，所有技术选型优先考虑**免费**和**本地部署**方案，旨在以最低成本快速验证项目可行性，完成功能开发和测试。

### 1.1 设计原则

- **零成本优先**: 所有服务和工具均使用免费方案
- **本地部署**: 核心服务本地运行，无需外部依赖
- **快速迭代**: 简化架构，专注功能开发
- **易于调试**: 本地环境便于问题排查和性能调优

### 1.2 适用场景

- 项目MVP开发
- 功能原型验证
- 单元测试和集成测试
- 性能测试和压力测试
- 团队内部演示

-- 创建识别记录表
CREATE TABLE recognition_records (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  file_id UUID NOT NULL,
  user_id UUID REFERENCES auth.users(id) ON DELETE SET NULL,
  table_data JSONB NOT NULL,
  confidence DECIMAL(3,2) DEFAULT 0.0,
  ocr_provider VARCHAR(50) DEFAULT 'tesseract',
  status VARCHAR(20) DEFAULT 'completed',
  metadata JSONB,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 创建索引
CREATE INDEX idx_recognition_file_id ON recognition_records(file_id);
CREATE INDEX idx_recognition_user_id ON recognition_records(user_id);
CREATE INDEX idx_recognition_created_at ON recognition_records(created_at DESC);

-- 创建自动更新时间戳的触发器
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_at = CURRENT_TIMESTAMP;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER update_recognition_records_updated_at
  BEFORE UPDATE ON recognition_records
  FOR EACH ROW
  EXECUTE FUNCTION update_updated_at_column();

-- 创建文件记录表
CREATE TABLE file_records (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  original_name VARCHAR(255) NOT NULL,
  file_path VARCHAR(500) NOT NULL,
  file_size BIGINT NOT NULL,
  mime_type VARCHAR(100) NOT NULL,
  width INTEGER,
  height INTEGER,
  status VARCHAR(20) DEFAULT 'active',
  created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
  expires_at TIMESTAMP WITH TIME ZONE
);

-- 创建索引
CREATE INDEX idx_file_records_status ON file_records(status);
CREATE INDEX idx_file_records_expires_at ON file_records(expires_at);

-- 启用Row Level Security（可选，用于用户隔离）
ALTER TABLE recognition_records ENABLE ROW LEVEL SECURITY;

-- 创建RLS策略
CREATE POLICY "用户可以查看自己的记录"
  ON recognition_records FOR SELECT
  USING (auth.uid() = user_id OR auth.uid() IS NULL);

CREATE POLICY "用户可以插入记录"
  ON recognition_records FOR INSERT
  WITH CHECK (auth.uid() = user_id OR auth.uid() IS NULL);

CREATE POLICY "用户可以更新记录"
  ON recognition_records FOR UPDATE
  USING (auth.uid() = user_id OR auth.uid() IS NULL);

CREATE POLICY "用户可以删除记录"

**数据操作示例**:

```typescript
// src/services/recognition.ts
import { supabase } from '@/lib/supabase'

export const createRecognitionRecord = async (data: {
  fileId: string
  userId?: string
  tableData: TableData
  confidence: number
  ocrProvider: string
  metadata: any
}) => {
  const { data: record, error } = await supabase
    .from('recognition_records')
    .insert({
      file_id: data.fileId,
      user_id: data.userId,
      table_data: data.tableData,
      confidence: data.confidence,
      ocr_provider: data.ocrProvider,
      metadata: data.metadata
    })
    .select()
    .single()

  if (error) throw error
  return record
}

export const getRecognitionRecord = async (id: string) => {
  const { data, error } = await supabase
    .from('recognition_records')
    .select('*')
    .eq('id', id)
    .single()

  if (error) throw error
  return data
}

export const getUserRecords = async (userId?: string, limit = 20) => {
  let query = supabase
    .from('recognition_records')
    .select('*')
    .order('created_at', { ascending: false })
    .limit(limit)

  if (userId) {
    query = query.eq('user_id', userId)
  }

  const { data, error } = await query

  if (error) throw error
  return data
}

export const updateRecognitionRecord = async (
  id: string,
  updates: Partial<{
    tableData: TableData
    confidence: number
    status: string
  }>
) => {
  const { data, error } = await supabase
    .from('recognition_records')
    .update(updates)
    .eq('id', id)
    .select()
    .single()

  if (error) throw error
  return data
}

export const deleteRecognitionRecord = async (id: string) => {
  const { error } = await supabase
    .from('recognition_records')
    .delete()
    .eq('id', id)

  if (error) throw error
}
```

// 监听识别记录的变化
export const subscribeToRecognition = (
  recordId: string,
  callback: (payload: any) => void
) => {
  const channel = supabase
    .channel(`recognition_${recordId}`)
    .on(
      'postgres_changes',
      {
        event: '*',
        schema: 'public',
        table: 'recognition_records',
        filter: `id=eq.${recordId}`
      },
      callback
    )
    .subscribe()

  return () => supabase.removeChannel(channel)
}

```python
# ocr_service.py
from flask import Flask, request, jsonify
from flask_cors import CORS
from paddleocr import PaddleOCR
import base64
import io
from PIL import Image

app = Flask(__name__)
CORS(app)

# 初始化OCR（CPU模式）
ocr = PaddleOCR(
    use_angle_cls=True,
    lang='ch',
    det=True,
    rec=True,
    use_gpu=False
)

@app.route('/api/recognize', methods=['POST'])
def recognize():
    data = request.json
    image_data = base64.b64decode(data['image'])
    image = Image.open(io.BytesIO(image_data))

    result = ocr.ocr(image, cls=True)
    
    # 解析结果
    table_data = []
    for line in result[0]:
        box = line[0]
        text_info = line[1]
        table_data.append({
            'text': text_info[0],
            'confidence': text_info[1]
        })
    
    return jsonify({
        'success': True,
        'data': {
            'tableData': table_data,
            'confidence': sum(d['confidence'] for d in table_data) / len(table_data)
        }
    })

if __name__ == '__main__':
    app.run(port=8000, debug=True)
```

**文件上传**:
```typescript
// src/routes/upload.ts
import express from 'express'
import multer from 'multer'
import path from 'path'
import { v4 as uuidv4 } from 'uuid'
import sharp from 'sharp'
import { config } from '../config'

const router = express.Router()

const storage = multer.diskStorage({
  destination: (req, file, cb) => {
    cb(null, config.uploadDir)
  },
  filename: (req, file, cb) => {
    const ext = path.extname(file.originalname)
    cb(null, `${uuidv4()}${ext}`)
  }
})

const upload = multer({
  storage,
  limits: { fileSize: config.maxFileSize },
  fileFilter: (req, file, cb) => {
    if (config.allowedTypes.includes(file.mimetype)) {
      cb(null, true)
    } else {
      cb(new Error('不支持的文件格式'))
    }
  }
})

router.post('/', upload.single('file'), async (req, res) => {
  try {
    if (!req.file) {
      return res.status(400).json({ error: '未上传文件' })
    }

    // 图片压缩
    const compressedPath = path.join(
      config.uploadDir,
      `compressed_${req.file.filename}`
    )
    await sharp(req.file.path)
      .resize(1920, 1080, { fit: 'inside' })
      .jpeg({ quality: 80 })
      .toFile(compressedPath)

    res.json({
      success: true,
      data: {
        fileId: req.file.filename,
        originalName: req.file.originalname,
        size: req.file.size,
        path: compressedPath
      }
    })
  } catch (error) {
    res.status(500).json({ error: '上传失败' })
  }
})

export default router
```

### 4.5 缓存（内存）

#### 内存缓存（简单方案）

```typescript
// src/cache/index.ts
class MemoryCache {
  private cache: Map<string, { data: any, expires: number }>

  constructor() {
    this.cache = new Map()
    this.startCleanup()
  }

  set(key: string, data: any, ttl: number = 3600) {
    const expires = Date.now() + ttl * 1000
    this.cache.set(key, { data, expires })
  }

  get(key: string) {
    const item = this.cache.get(key)
    if (!item) return null

    if (Date.now() > item.expires) {
      this.cache.delete(key)
      return null
    }

    return item.data
  }

  delete(key: string) {
    this.cache.delete(key)
  }

  clear() {
    this.cache.clear()
  }

  private startCleanup() {
    setInterval(() => {
      const now = Date.now()
      for (const [key, item] of this.cache.entries()) {
        if (now > item.expires) {
          this.cache.delete(key)
        }
      }
    }, 60000) // 每分钟清理一次
  }
}

export default new MemoryCache()
```
